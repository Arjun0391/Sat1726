Operators in Mongo Db:
// To fetch the Unit Price Less than 2000 Values
db.Sales.find({UnitPrice:{$lt:2000}})
// To fetch the Sales Amount Greater than 100000 Values
db.Sales.find({SalesAmount:{$gt:10000}})
// To Fetch the greater than or Equal to Qunatity Values
db.Sales.find({Quantity:{$gte:5}})
//To fetch SalesAmount the Less than 20000
db.Sales.find({SalesAmount:{$lte:2000}})
// To fetch the Unit price Equal to 1000
db.Sales.find({UnitPrice:{$eq:1000}})
// To fetch Quantity is not equal 4
db.Sales.find({Quantity:{$ne:4}})


// Group by functions
db.Sales.aggregate([{$group:{_id:"$ProductName",TotalNUmberofQuantity:{$sum:"$Quantity"}}}])

//To fetch the average price
db.Sales.aggregate([{$group:{_id:"$ProductName",AveragePrice:{$avg:"$UnitPrice"}}}])
// To fetch the count values
db.Sales.aggregate([{$group:{_id:"$ProductName",Count:{$count:{}}}}])
// NULL Keyword is for retrive all the count
db.Sales.aggregate([{$group:{_id:"NULL",Count:{$count:{}}}}])

// To update value if one document
db.Sales.updateOne({ ProductName: "GraphicsCard" },{ $set: { UnitPrice: 6000 } })
// To update many value of more than one document
db.Sales.updateMany({ProductName:"MousePad"},{$set:{Quantity:50}})

//Upseret is a combination of update & insert, if there is a matching count, it will modify the value if no matching field it will add a new document with specified value

db.Sales.updateOne({ProductName:"ColorBox"},{$set:{UnitPrice:700}},{upsert:true})

db.Sales.updateMany({ProductName:"ColorBox"},{$set:{UnitPrice:700,Quantity:5,SalesAmount:3500}},{upsert:true})

// to sort the column value in ascending order
db.Sales.find().sort({UnitPrice:1})
//to sort the column value in Descending order
db.Sales.find().sort({UnitPrice:-1})
